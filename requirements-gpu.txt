# Real LLM + STT/TTS deps -- install these on the Runpod GPU box in
# addition to requirements.txt, then set MOCK_MODE=false (or unset it).

# file; `pip install -r requirements.txt -r requirements-gpu.txt` is
# sufficient on its own -- no separate manual `pip install vllm` step.

# System dependency: ffmpeg (faster-whisper/ctranslate2 shells out to it to
# decode whatever audio format the browser recorded -- e.g. webm/opus).
#   apt-get install -y ffmpeg

torch
torchaudio
faster-whisper
